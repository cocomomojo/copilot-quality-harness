  12export function isValidURL(url) {
  13  // Check for empty or null input
  14  if (url === null || url === undefined || url === '') {
  15    return { valid: false, error: 'URL cannot be empty' };
  16  }
  17
  18  // Check if URL is a string
  19  if (typeof url !== 'string') {
  20    return { valid: false, error: 'URL must be a string' };
  21  }
  22
  23  // Check length limit (2048 characters per RFC 3986)
  24  if (url.length > 2048) {
  25    return { valid: false, error: 'Invalid URL format' };
  26  }
  27
  28  // Check if URL starts with allowed protocols
  29  if (!url.startsWith('http://') && !url.startsWith('https://')) {
  30    return { valid: false, error: 'Invalid URL format' };
  31  }
  32
  33  // Try to parse as URL
  34  try {
  35    // This will throw if URL is malformed
  36    new URL(url);
